model PricingTier {
  // ... existing fields
  
  // Новые поля
  includesDelivery     Boolean   @default(false)
  includedSmsCount     Int       @default(0)      // 0, 50, или -1 (безлимит)
  smsOveragePrice      Float     @default(0.10)   // Цена за SMS сверх лимита
  maxCustomers         Int?                       // Лимит базы клиентов

model Subscription {
  // ... existing fields

  addons               String[]  @default([])     // ["delivery", "analytics", "customers_pro"]

}

### 2. **Лимиты базы клиентов** (3 дня)
```javascript
// Middleware для проверки лимита
const checkCustomerLimit = async (req, res, next) => {
  const restaurant = await getRestaurant();
  const subscription = restaurant.subscription;
  const tier = subscription.pricingTier;
  
  const customerCount = await prisma.customer.count({
    where: { registeredRestaurantId: restaurant.id }
  });
  
  if (tier.maxCustomers && customerCount >= tier.maxCustomers) {
    return res.status(402).json({ 
      error: 'Customer limit reached. Please upgrade your plan.'
    });
  }
  
  next();
};
```

### 3. **SMS биллинг** (1 неделя)
```javascript
// Функция отправки SMS с подсчетом
const sendSmsWithBilling = async (phone, code, restaurantId) => {
  const subscription = await getSubscription(restaurantId);
  const tier = subscription.pricingTier;
  
  // Проверяем лимит
  if (tier.includedSmsCount !== -1) { // Не безлимит
    if (subscription.smsUsedThisMonth >= tier.includedSmsCount) {
      // Начисляем overage charge
      await createInvoiceItem({
        subscriptionId: subscription.id,
        description: `SMS overage (${phone})`,
        amount: tier.smsOveragePrice
      });
    }
  }
  
  // Увеличиваем счетчик
  await prisma.subscription.update({
    where: { id: subscription.id },
    data: { smsUsedThisMonth: { increment: 1 } }
  });
  
  // Отправляем SMS
  await twilioClient.messages.create({...});
};
```
